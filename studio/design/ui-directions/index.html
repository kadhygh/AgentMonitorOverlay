DropTableValidator.cs — mining-a
public static class DropTableValidator
{
    // 概率和允许的误差
    const float Epsilon = 0.0001f;

    public static IEnumerable<string> FindInvalid(DropTable[] tables)
    {
        foreach (var table in tables)
        {
            var sum = table.Entries.Sum(e => e.Weight);
            if (Mathf.Abs(sum - 1f) > Epsilon)
                yield return $"{table.name}: {sum:0.000}";
        }
    }
}
